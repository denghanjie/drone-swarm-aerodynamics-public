def require(ok, code):
    report['checks'] += 1
    if not ok:
        raise RuntimeError(code)
    if time.monotonic() >= deadline:
        raise TimeoutError('Saved review deadline')

def digest(data):
    return hashlib.sha256(data).hexdigest()

def safe_file(root, relative):
    p = Path(relative)
    require(isinstance(relative, str) and relative and not p.is_absolute()
            and all(x not in ['', '.', '..'] for x in relative.split('/')), 'relative-evidence-path')
    target = root / p
    require(target.resolve() == target and stat.S_ISREG(target.lstat().st_mode), 'regular-contained-file')
    return target

def read_path(target, cap=1048576):
    require(stat.S_ISREG(target.lstat().st_mode) and target.stat().st_size <= cap, 'bounded-regular-input')
    with target.open('rb') as handle:
        data = handle.read(cap + 1)
    require(len(data) <= cap, 'bounded-input-growth')
    return data

def read(relative, cap=1048576):
    return read_path(safe_file(BASE, relative), cap)

def load(relative):
    return json.loads(read(relative))

def check_pin(root, pin):
    require(set(pin) == {'path', 'bytes', 'sha256'} and type(pin['bytes']) is int
            and 0 <= pin['bytes'] <= 16777216 and isinstance(pin['sha256'], str)
            and len(pin['sha256']) == 64, 'pin-schema')
    target = safe_file(root, pin['path'])
    require(target.stat().st_size == pin['bytes'], 'pinned-size')
    count = 0
    hashed = hashlib.sha256()
    with target.open('rb') as handle:
        while True:
            part = handle.read(65536)
            if not part:
                break
            count += len(part)
            require(count <= pin['bytes'], 'pinned-growth')
            hashed.update(part)
    require(count == pin['bytes'] and hashed.hexdigest() == pin['sha256'], 'pinned-bytes')

def check_pins(root, pins):
    require(isinstance(pins, list) and len(pins) <= 256
            and len({p['path'] for p in pins}) == len(pins), 'unique-pin-inventory')
    for pin in pins:
        check_pin(root, pin)

def names(relative):
    directory = BASE / relative
    require(directory.is_dir() and not directory.is_symlink(), 'regular-evidence-directory')
    result = []
    for item in directory.iterdir():
        require(len(result) < 256 and stat.S_ISREG(item.lstat().st_mode), 'flat-evidence-inventory')
        result.append(item.name)
    return sorted(result)

def private_files(manifest):
    locator = Path(os.environ['M31_RUNTIME_REVIEW_PRIVATE_LOCATIONS'])
    require(locator.is_absolute(), 'absolute-private-locator')
    locator_data = read_path(locator, 16384)
    require(digest(locator_data) == manifest['privateLocatorSHA256'], 'private-locator-binding')
    locations = json.loads(locator_data)
    require(len(locations) == len(manifest['privateGroups']) == 2, 'two-private-groups')
    output = {}
    for expected, actual in zip(manifest['privateGroups'], locations):
        require(actual['files'] == expected['files'], 'private-inventory-binding')
        root = Path(actual['path'])
        require(root.is_absolute() and root.is_dir() and not root.is_symlink(), 'private-directory')
        root = root.resolve()  # macOS /var is an alias; final files must be canonical.
        require(sorted(x.name for x in root.iterdir()) == sorted(x['name'] for x in expected['files']), 'private-exact-files')
        output[expected['label']] = {}
        for item in expected['files']:
            target = safe_file(root, item['name'])
            data = read_path(target, 65536)
            require(len(data) == item['bytes'] and digest(data) == item['sha256'], 'private-pinned-bytes')
            output[expected['label']][item['name']] = data
    return output
